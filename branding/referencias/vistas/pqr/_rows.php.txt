 function pqr_rows_subject_view($raw) {
  $raw = (string)$raw;
  // QA-only prefix is metadata; keep the complete source in the HTML title.
  if (preg_match('/^\[(QA[A-Z0-9_-]{6,48})\]\s*(\S.*)$/iu', $raw, $matches)) {
   return ['text' => $matches[2], 'qa' => $matches[1], 'full' => $raw];
  }
  return ['text' => $raw, 'qa' => '', 'full' => $raw];
 }
 function pqr_rows_compact_priority($label) {
  $label = (string)$label;
  // Only abbreviate the known full overdue expression, never recalculate legal terms.
  if (preg_match('/^Vencida hace\s+(\d+)\s+d[ií]a(?:s)?\s+h[aá]bil(?:es)?$/iu', trim($label), $matches)) {
   return 'Vencida · '.$matches[1].' d. háb.';
  }
  // Un resultado de 0 días hábiles no significa que la fecha de vencimiento sea hoy.
  if (preg_match('/^Faltan\s+0\s+d[ií]as?\s+h[aá]bil(?:es)?$/iu', trim($label))) {
   return 'Sin días hábiles';
  }
  return $label;
 }
