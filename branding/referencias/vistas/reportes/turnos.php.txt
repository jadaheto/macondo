document.addEventListener('DOMContentLoaded', function () {
  const endpoints = {
    datatable: '<?= site_url('reportes/api_turnos_datatable') ?>',

    crearExportacion: '<?= site_url('reportes/api_crear_exportacion') ?>',
    procesarExportacion: '<?= site_url('reportes/api_procesar_exportacion') ?>',
    exportaciones: '<?= site_url('reportes/api_exportaciones') ?>',
    descargar: '<?= site_url('reportes/descargar') ?>',
    guardarFiltro: '<?= site_url('reportes/api_guardar_filtro') ?>',
    filtrosGuardados: '<?= site_url('reportes/api_filtros_guardados') ?>'
  };

  const modalFiltros = new bootstrap.Modal(document.getElementById('modalFiltros'));
  const modalDetalle = new bootstrap.Modal(document.getElementById('modalDetalle'));
  const filterForm = document.getElementById('formFiltros');
  let filtros = {};
  let filtrosGuardados = [];

  function formDataObject() {
    const data = {};
    $('#formFiltros').serializeArray().forEach(item => { data[item.name] = item.value; });
    return data;
  }

  function syncDraft(data) {
    filterForm.reset();
    Array.from(filterForm.elements).forEach(control => {
      if (control.name && Object.prototype.hasOwnProperty.call(data, control.name)) control.value = data[control.name] ?? '';
    });
    updatePresets();
    document.getElementById('dashFilterError').hidden = true;
  }

  function updatePresets() {
    const draft = formDataObject();
    document.querySelectorAll('[data-period]').forEach(button => {
      const range = TurnosDashboard.periodo(button.dataset.period);
      button.setAttribute('aria-pressed', String(draft.fecha_desde === range.fecha_desde && draft.fecha_hasta === range.fecha_hasta));
    });
  }

  function setDraftPeriod(value) {
    const range = TurnosDashboard.periodo(value);
    filterForm.elements.fecha_desde.value = range.fecha_desde;
    filterForm.elements.fecha_hasta.value = range.fecha_hasta;
    updatePresets();
  }

  // Bound the very first table request; saved filters never override startup.
  setDraftPeriod('7');
  filtros = formDataObject();
  Array.from(filterForm.elements).forEach(control => {
    if (control.name && !control.id) control.id = 'reportesFiltro_' + control.name;
    const label = control.parentElement.querySelector('label');
    if (label && control.id) label.htmlFor = control.id;
  });

  function renderChips() {
    const labels = [];
    const data = filtros;
    Object.keys(data).forEach(function (key) {
      if (!data[key]) return;
      if (key === 'fecha_desde' || key === 'fecha_hasta') return;
      if (key === 'fecha_campo' && data[key] === 'creado_en') return;
      const control = document.querySelector('#formFiltros [name="' + key + '"]');
      const label = control?.parentElement.querySelector('label')?.textContent || key.replace(/_/g, ' ');
      const value = control?.tagName === 'SELECT' ? Array.from(control.options).find(option => option.value === String(data[key]))?.textContent ?? data[key] : data[key];
      labels.push(`<button type="button" class="filter-chip" data-remove="${escapeHtml(key)}" aria-label="Quitar filtro ${escapeHtml(label)}">${escapeHtml(label)}: ${escapeHtml(value)} ×</button>`);
    });
    $('#chipsFiltros').html(labels.length ? labels.join('') : '<span class="dash-summary-default">Todos los servicios · Todos los destinos</span>');
    $('#dashFilterCount').text(labels.length).attr('aria-label',labels.length+' filtros adicionales al periodo').prop('hidden',labels.length===0);
    const dateLabel=value=>new Intl.DateTimeFormat('es-CO',{timeZone:'America/Bogota',day:'numeric',month:'short',year:'numeric'}).format(new Date(value+'T12:00:00-05:00'));
    $('#dashPeriodSummary').text(data.fecha_desde===data.fecha_hasta ? (data.fecha_desde===TurnosDashboard.today()?'Hoy · ':'')+dateLabel(data.fecha_desde) : dateLabel(data.fecha_desde)+' — '+dateLabel(data.fecha_hasta));
  }

  function escapeHtml(value) {
    return String(value || '').replace(/[&<>"']/g, function (m) {
      return ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#039;'})[m];
    });
  }

  function estadoPill(value) {
    const safe = escapeHtml(value || '--');
    return `<span class="estado-pill ${safe}">${safe.replace('_', ' ')}</span>`;
  }

  function renderFecha(data, type) {
    return type === 'display' || type === 'filter'
      ? escapeHtml(TurnosFecha.fechaHora(data, {segundos: false}))
      : data;
  }

    searching: true,

      { data:'prioridad', render: data => data === 'Normal' ? '<span class="text-muted">Normal</span>' : `<span class="badge bg-success-subtle text-success">${escapeHtml(data)}</span>` },

  });

  const dashboard = TurnosDashboard.create({endpoint: endpoints.dashboard, catalogos: <?= json_encode($catalogos, JSON_HEX_TAG|JSON_HEX_AMP|JSON_HEX_APOS|JSON_HEX_QUOT) ?>,
    filter: function(key, value) { filtros = {...filtros, [key]: String(value)}; refrescarTodo(); }});
  function cargarResumen(force) { dashboard.load(filtros,force); }

  function cargarExportaciones() {
    $.getJSON(endpoints.exportaciones, function (resp) {
      const items = resp && resp.ok ? resp.data : [];
      if (!items.length) {
        $('#listaExportaciones').html('<div class="text-muted">Aun no hay descargas.</div>');
        return;
      }
      $('#listaExportaciones').html(items.map(function (item) {
        const ready = item.estado === 'LISTO';
        const url = `${endpoints.descargar}/${item.id}`;
        return `<div class="download-item">
          <div class="d-flex justify-content-between gap-2 mb-1">
            <strong>${escapeHtml(item.formato)}</strong>
            <span class="download-status ${escapeHtml(item.estado)}">${escapeHtml(item.estado)}</span>
          </div>
          <div class="text-muted small">${escapeHtml(TurnosFecha.fechaHora(item.creado_en, {segundos: true}))}</div>
          <div class="small">${parseInt(item.total_registros || 0, 10)} registro(s)</div>
          ${ready ? `<a class="btn btn-sm btn-primary mt-2 w-100" href="${url}"><i class="ri-download-2-line me-1"></i> Descargar</a>` : ''}
          ${item.estado === 'ERROR' ? `<div class="text-danger small mt-2">${escapeHtml(item.mensaje || 'Error')}</div>` : ''}
        </div>`;
      }).join(''));
    });
  }

  function cargarFiltrosGuardados() {
    $.getJSON(endpoints.filtrosGuardados, function (resp) {
      filtrosGuardados = resp && resp.ok ? resp.data : [];
      const options = ['<option value="">Selecciona un filtro guardado</option>'].concat(filtrosGuardados.map(function (item) {
        return `<option value="${item.id}">${escapeHtml(item.nombre)}</option>`;
      }));
      $('#selectFiltrosGuardados').html(options.join(''));
    });
  }

  function refrescarTodo(force) {
    renderChips();
    cargarResumen(force === true);
    table.ajax.reload();
  }

  $('#btnToggleFiltros').on('click', () => modalFiltros.show());
  document.getElementById('modalFiltros').addEventListener('show.bs.modal', function () {
    syncDraft(filtros);
    document.getElementById('btnAplicarFiltros').disabled = false;
    $('#selectFiltrosGuardados').val('');

  document.getElementById('modalFiltros').addEventListener('shown.bs.modal', function () {

  });
  document.getElementById('modalFiltros').addEventListener('hidden.bs.modal', function () {

    document.getElementById('btnToggleFiltros').focus();
  });

  function applyFilters() {
    const applyButton = document.getElementById('btnAplicarFiltros');
    if (applyButton.disabled) return;
    const draft = formDataObject(), error = document.getElementById('dashFilterError');
    const validDate = value => /^\d{4}-\d{2}-\d{2}$/.test(value) &&
      !Number.isNaN(Date.parse(value + 'T12:00:00Z')) &&
      new Date(value + 'T12:00:00Z').toISOString().slice(0,10) === value;
    if (!validDate(draft.fecha_desde) || !validDate(draft.fecha_hasta) || draft.fecha_desde > draft.fecha_hasta) {
      error.hidden = false;
      error.textContent = 'Indica fechas válidas. Desde debe ser anterior o igual a Hasta.';
      filterForm.elements.fecha_desde.focus();
      return;
    }
    if (!filterForm.reportValidity()) return;
    applyButton.disabled = true;
    filtros = draft;
    modalFiltros.hide();
    refrescarTodo();
  }
  $('#btnAplicarFiltros').on('click', applyFilters);
  $('#btnLimpiarFiltros').on('click', function () {
    syncDraft({fecha_campo:'creado_en', ...TurnosDashboard.periodo('7')});
    $('#selectFiltrosGuardados').val('');
  });

    refrescarTodo();

  $('#btnGuardarFiltro').on('click', async function () {
    const applied = {...filtros};
    const result = await Swal.fire({ title:'Guardar filtro', input:'text', inputLabel:'Nombre del filtro', showCancelButton:true, confirmButtonText:'Guardar' });
    if (!result.isConfirmed || !result.value) return;
    $.post(endpoints.guardarFiltro, Object.assign({}, applied, {nombre_filtro: result.value}), function (resp) {
      if (resp && resp.ok) TurnosAvisos.mostrar('Listo', 'Filtro guardado.', 'success');
      else Swal.fire('Aviso', (resp && resp.msg) || 'No fue posible guardar.', 'warning');
      if (resp && resp.ok) cargarFiltrosGuardados();
    }, 'json');
  });

  $('#btnCargarFiltroGuardado').on('click', function (event) {
    event.preventDefault();
    const id = parseInt($('#selectFiltrosGuardados').val(), 10);
    const item = filtrosGuardados.find(row => parseInt(row.id, 10) === id);
    if (!item) return;
    try {
      const data = typeof item.filtros === 'string' ? JSON.parse(item.filtros) : item.filtros;
      if (!data || typeof data !== 'object' || Array.isArray(data)) throw new Error('invalid');
      syncDraft({fecha_campo:'creado_en', ...TurnosDashboard.periodo('7'), ...data});
    } catch (_error) {
      $('#dashFilterError').prop('hidden',false).text('No se pudo cargar este filtro guardado.');
    }
  });

  $('[data-export]').on('click', function () {
    const formato = $(this).data('export');
    $.post(endpoints.crearExportacion, Object.assign({}, filtros, {formato}), function (resp) {
      if (!resp || !resp.ok) {
        Swal.fire('Aviso', resp && resp.msg ? resp.msg : 'No fue posible crear la exportacion.', 'warning');
        return;
      }
      TurnosAvisos.mostrar('Preparando archivo', 'Lo agregamos al centro de descargas.', 'info');
      procesarExportacion(resp.id);
    }, 'json');
  });

  function procesarExportacion(id) {
    $.post(endpoints.procesarExportacion, {id}, function () {
      cargarExportaciones();
    }, 'json').fail(function () {
      cargarExportaciones();
    });
  }

    $('#detalleContenido').html('<div class="text-center text-muted py-4">Cargando detalle...</div>');

    $.getJSON(endpoints.detalle, {id}, function (resp) {

          <div>${estadoPill(t.estado)} ${t.prioridad == 1 ? '<span class="badge bg-success-subtle text-success ms-1">Prioritario</span>' : ''}</div>
        </div>
        <div class="detail-grid mb-4">
          ${detailBox('Dependencia', t.nombre_dependencia)}
          ${detailBox('Modulo', t.nombre_modulo || 'Sin modulo')}
          ${detailBox('Tipo persona', t.tipo_persona)}
          ${detailBox('Documento', `${t.tipo_id || ''} ${t.num_id || ''}`)}
          ${detailBox('Celular', t.celular)}
          ${detailBox('Correo', t.email)}
          ${detailBox('Empresa', t.empresa_nombre)}
          ${detailBox('Gestor', t.gestor_nombre)}
          ${detailBox('Creado', TurnosFecha.fechaHora(t.creado_en, {segundos: true}))}
          ${detailBox('Llamado', TurnosFecha.fechaHora(t.llamado_en, {segundos: true}))}
          ${detailBox('Inicio atencion', TurnosFecha.fechaHora(t.iniciado_en, {segundos: true}))}
          ${detailBox('Finalizado', TurnosFecha.fechaHora(t.finalizado_en, {segundos: true}))}
          ${detailBox('Espera', TurnosDashboard.duracion(t.espera_minutos))}
          ${detailBox('Atencion', TurnosDashboard.duracion(t.atencion_minutos))}
          ${detailBox('Prioridad', t.prioridad_tipo || 'Normal')}
          ${detailBox('Observaciones', t.observaciones)}
        </div>
        <h6 class="mb-3">Trazabilidad</h6>
        ${eventos.length ? eventos.map(e => `<div class="timeline-event"><strong>${escapeHtml(e.evento)}</strong><div class="text-muted small">${escapeHtml(TurnosFecha.fechaHora(e.creado_en, {segundos: true}))} · ${escapeHtml(e.usuario_nombre || 'Sistema')}</div><div>${escapeHtml(e.observacion || '')}</div></div>`).join('') : '<div class="text-muted">Sin eventos registrados.</div>'}
      `);
    });
  });

  function detailBox(label, value) {
    return `<div class="detail-box"><span>${escapeHtml(label)}</span><strong>${escapeHtml(value || '--')}</strong></div>`;
  }

  renderChips();
  cargarResumen();

  cargarFiltrosGuardados();
});
