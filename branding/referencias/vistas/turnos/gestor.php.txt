                }
                // Presentation clock is explicit Bogotá; duration keeps its existing parser.
                const serverClockBogota = TurnosFecha.parse(resp.server_now);
                if (serverClockBogota) {
                    serverClockBogotaOffsetMs = serverClockBogota.getTime() - Date.now();
                }

                turnoActual = resp.turno_actual || null;
                puedeTransferir = resp.puede_transferir !== false;
                modulos = resp.modulos || [];
                modulosDestino = resp.modulos_destino || [];
                dependencias = resp.dependencias || [];
                $('#dependenciasModuloGestor').text('Dependencias que atiende: ' + ((resp.dependencias_modulo || []).map(function(d){return d.nombre;}).join(' · ') || 'Sin dependencias configuradas'));
                disponibilidadModulo = resp.disponibilidad || { disponible: true, estado: 'DISPONIBLE', label: 'Disponible' };
                renderTurnoActual();
                renderCola(resp.cola || []);
                llenarSelectsTransferencia();
                actualizarSelectorModulos();
                renderDisponibilidadModulo();
                window.dispatchEvent(new CustomEvent('turnos:gestor-estado',{detail:{avisos:resp.avisos_cola || null}}));
            })
            .fail(function (xhr) {
                window.dispatchEvent(new CustomEvent('turnos:gestor-invalido',{detail:{revocado:xhr.status===401 || xhr.status===403}}));
                if(!soloAvisos)Swal.fire('Error', 'No fue posible cargar el estado del gestor.', 'error');
            }).always(function(){estadoPeticion=null;});
    }

    function actualizarSelectorModulos() {
        if (!modulos.length) return;
        const options = modulos.map(function (m) {
            const disp = m.disponibilidad || {};
            const ocupadoOtro = !!m.ocupado && !m.ocupado_por_mi;
            let etiqueta = ocupadoOtro ? ' · Ocupado' : '';
            etiqueta += disp.disponible === false ? ' · No disponible' : '';
            return `<option value="${m.id}" ${parseInt(m.id, 10) === moduloId ? 'selected' : ''} ${ocupadoOtro ? 'disabled' : ''}>${escapeHtml(rotuloDestino(m) + (m.ocupado_por_nombre ? ' · '+m.ocupado_por_nombre : '') + etiqueta)}</option>`;
        }).join('');
        // Do not replace focused options on every poll: keyboard navigation must survive updates.
        const selector = document.getElementById('selectorModuloGestor');
        if (document.activeElement !== selector && selector.dataset.optionsMarkup !== options) {
            selector.innerHTML = options;
            selector.dataset.optionsMarkup = options;
        }
        $('#btnLiberarPuestoTexto').text(esOficinaActual() ? 'Liberar puesto' : 'Liberar módulo');
    }

        }

        const html = cola.map(function (t, idx) {
            const prioridad = parseInt(t.prioridad, 10) > 0;
            const moduloDestino = t.modulo_id ? rotuloDestino(t) : 'Cola general';
            return `
                <div class="list-group-item gt-queue-row">
                    <div class="gt-queue-ticket">${escapeHtml(codigoTurno(t))}</div>

                      <div class="gt-queue-name">${escapeHtml(nombrePersona(t))}</div>

    function llenarSelectsTransferencia() {
        const anterior = $('#transferDependencia').val();
        let depOptions = '<option value="">Conservar dependencia actual</option>';
        depOptions += dependencias.map(function (d) {
            return `<option value="${d.id}">${escapeHtml(d.nombre)}</option>`;
        }).join('');
        const selector = document.getElementById('transferDependencia');
        if (document.activeElement !== selector && selector.dataset.optionsMarkup !== depOptions) {
            selector.innerHTML = depOptions;

    }

    function filtrarModulosTransferencia() {
        const destino = Number($('#transferDependencia').val() || (turnoActual && turnoActual.dependencia_id) || 0);
        const previo = $('#transferModulo').val();
        let moduloOptions = '<option value="">Selecciona un módulo o puesto de oficina con gestor disponible</option>';
        moduloOptions += modulosDestino.filter(function(m){return (m.dependencias || []).some(function(d){return Number(d.id)===destino;});}).map(function (m) {
            const disp = m.disponibilidad || {};
            const etiqueta = disp.disponible === false ? ' · No disponible' : '';
            const disabled = !m.ocupado || disp.disponible !== true || Number(m.id)===moduloId ? 'disabled' : '';
            return `<option value="${m.id}" ${disabled}>${escapeHtml(rotuloDestino(m) + (m.ocupado_por_nombre ? ' · '+m.ocupado_por_nombre : '') + etiqueta)}</option>`;
        }).join('');
        const selector = document.getElementById('transferModulo');
        const compatibles=modulosDestino.filter(m=>(m.dependencias||[]).some(d=>Number(d.id)===destino));
        const disponibles=compatibles.filter(m=>m.ocupado && m.disponibilidad?.disponible===true && Number(m.id)!==moduloId);
        $('#transferAyudaDestino').text(!compatibles.length?'No hay puestos que atiendan este servicio. Si vas a otra área, cambia primero el servicio.':(!disponibles.length?'Los puestos compatibles están vacíos, pausados, cerrados o son tu puesto actual. Solicita disponibilidad al destino.':'Selecciona el puesto y gestor de destino.'));
        if (document.activeElement === selector) return;
        if (selector.dataset.optionsMarkup !== moduloOptions) {
            selector.innerHTML = moduloOptions;
            selector.dataset.optionsMarkup = moduloOptions;
        }
        if ($('#transferModulo option').toArray().some(function(o){return o.value===previo&&!o.disabled;})) $('#transferModulo').val(previo);
    }
    $('#transferDependencia').on('change', filtrarModulosTransferencia);

    function enviarOperacion(url, payload, callback) {
        if (accionEnCurso) return $.Deferred().reject({status:409, responseJSON:{msg:'Ya hay una acción en proceso.'}}).promise();
        accionEnCurso = true;
        window.dispatchEvent(new CustomEvent('turnos:gestor-operacion',{detail:{enCurso:true}}));
        estadoVersion++;
        const controles = $('.gestor-turnos button, .gestor-turnos select, #modalTransferir button, #modalDisponibilidad button');
        controles.each(function () { $(this).data('turnosDisabledAnterior', this.disabled).prop('disabled', true); });
        const peticion = $.ajax({url:url, type:'POST', data:payload, dataType:'json', timeout:10000});
        peticion.done(callback).always(function () {
            accionEnCurso = false;
            window.dispatchEvent(new CustomEvent('turnos:gestor-operacion',{detail:{enCurso:false}}));
            controles.each(function () { $(this).prop('disabled', !!$(this).data('turnosDisabledAnterior')); });
            cargarEstado();
        });
        return peticion;
    }

    function postAccion(url, payload, successMsg, options) {
        if (accionEnCurso) return;
        options = options || {};
        enviarOperacion(url, payload, function (resp) {
            if (resp && resp.ok) {
                if (resp.turno) {
                    turnoActual = resp.turno;
                    renderTurnoActual();
                }
                if (options.silent) {
                    cargarEstado();
                    return;
                }
                TurnosAvisos.mostrar('Listo', resp.msg || successMsg || 'Acción realizada.', 'success');
                cargarEstado();
            } else {
                if (resp && resp.disponibilidad) {
                    disponibilidadModulo = resp.disponibilidad;
                    renderDisponibilidadModulo();
                }
                Swal.fire('Atención', (resp && resp.msg) || 'No fue posible completar la acción.', 'warning');
                cargarEstado();
            }
        }).fail(function (xhr) {
            const resp = xhr.responseJSON;
            Swal.fire(xhr.status === 409 ? 'Atención' : 'Error', (resp && resp.msg) || 'No fue posible comunicarse con el servidor.', xhr.status === 409 ? 'warning' : 'error');
        });
    }

    $('.accion-turno').on('click', function () {
        if (!turnoActual) {
            TurnosAvisos.mostrar('Sin turno', 'No hay turno actual para gestionar.', 'info');
            return;
        }

        const accion = $(this).data('accion');
        const payloadBase = { turno_id: turnoActual.id };

        if (accion === 'rellamar') {
            postAccion(endpoints.rellamar, payloadBase, 'Turno rellamado correctamente.');
            return;
        }

        if (accion === 'iniciar') {
            postAccion(endpoints.iniciar, payloadBase, 'Atención iniciada correctamente.', { silent: true });
            return;
        }

        if (accion === 'finalizar') {
            pedirObservacion('Finalizar turno', false, function (obs) {
                postAccion(endpoints.finalizar, { turno_id: turnoActual.id, observacion: obs }, 'Turno finalizado.');
            });
            return;
        }

        if (accion === 'ausente') {
            pedirObservacion('Marcar como ausente', false, function (obs) {
                postAccion(endpoints.ausente, { turno_id: turnoActual.id, observacion: obs }, 'Turno marcado como ausente.');
            });
            return;
        }

        if (accion === 'cancelar') {
            pedirObservacion('Cancelar turno', true, function (obs) {
                postAccion(endpoints.cancelar, { turno_id: turnoActual.id, observacion: obs }, 'Turno cancelado.');
            });
        }
    });

    $('#btnAbrirTransferencia').on('click', function () {
        if (!turnoActual) {
            TurnosAvisos.mostrar('Sin turno', 'No hay turno actual para transferir.', 'info');
            return;
        }

        $('#transferTurnoId').val(turnoActual.id);
        $('#transferTurnoId').data('etapa',Number(turnoActual.etapa)||1);
        $('#transferDependencia').val(turnoActual.dependencia_id || '');
        filtrarModulosTransferencia();
        $('#transferModulo').val('');
        $('#transferObservacion').val('');
        modalTransferir.show();
    });

    $('#btnConfirmarTransferencia').on('click', function () {
        if(!$('#transferModulo').val()){Swal.fire('Destino','Selecciona un módulo o puesto de oficina compatible con gestor disponible.','warning');return;}
        const payload = {
            turno_id: $('#transferTurnoId').val(),
            etapa_esperada: $('#transferTurnoId').data('etapa'),
            puesto_destino_id: $('#transferModulo').val(),
            dependencia_destino_id: $('#transferDependencia').val(),
            observacion: $('#transferObservacion').val()
        };

        enviarOperacion(endpoints.transferir, payload, function (resp) {
            if (resp && resp.ok) {
                modalTransferir.hide();
                TurnosAvisos.mostrar('Listo', resp.msg || 'Turno transferido a la cola de destino.', 'success');
                cargarEstado();
            } else {
                Swal.fire('Atención', (resp && resp.msg) || 'No fue posible transferir el turno.', 'warning');
            }
        }, 'json').fail(function (xhr) {
            Swal.fire('Atención', (xhr.responseJSON && xhr.responseJSON.msg) || 'No fue posible transferir el turno.', 'warning');
        });
    });

    cargarEstado();
    window.addEventListener('turnos:actualizar-estado',cargarEstado);
    setInterval(actualizarDuracionAtencion, 1000);
    setInterval(function(){
        if(Date.now()-ultimaCargaCompleta>=15000)cargarEstado(false);
        else if(document.visibilityState==='visible')cargarEstado(true);
    }, 2000);
});
